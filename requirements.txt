"""
Research Copilot logic executing grounding structures and a number guard check.
Strictly isolates verified outputs, preventing hallucinations from leaking out.
"""
import re
from typing import Dict, List, Any, Set
import streamlit as st
from facts import extract_market_facts

def extract_numbers_from_text(text: str) -> Set[str]:
    found = re.findall(r"\d+\.\d+|\d+", text)
    return set(found)

def apply_number_guard(llm_response: str, verified_facts: List[Dict[str, Any]]) -> str:
    fact_tokens: Set[str] = set()
    for f in verified_facts:
        fact_tokens.update(extract_numbers_from_text(f["text"]))
    words = llm_response.split(" ")
    sanitized_words = []
    for word in words:
        numbers_in_word = extract_numbers_from_text(word)
        flagged = False
        for num in numbers_in_word:
            if num not in fact_tokens and len(num) > 1: 
                flagged = True
                break
        if flagged:
            sanitized_words.append("[HALLUCINATION_FLAG]")
        else:
            sanitized_words.append(word)
    return " ".join(sanitized_words)

    gemini_key = st.secrets.get("GEMINI_API_KEY", None)
    anthropic_key = st.secrets.get("ANTHROPIC_API_KEY", None)
    if not gemini_key and not anthropic_key:
        return ("INTERPRETATION: Based purely on the computed technical profile, the market reveals localized trend patterns. "
                "ASSUMPTION: We lean on the assumption that momentum signals persist across short horizons absent macro shocks. "
                "\n[Note: Live AI reasoning layers are currently disconnected. Displaying core quantitative analysis framework inputs].")
    return ("INTERPRETATION: The technical layout indicates a localized consolidation breakout pattern. "
